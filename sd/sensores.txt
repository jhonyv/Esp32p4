# =====================================================================
#  sensores.txt  -  de donde lee la pantalla cada valor
#  Copiar este archivo a la raiz de la SD (al lado de la carpeta dashboards).
#  Sin este archivo, todos los valores salen SIMULADOS (como antes).
# =====================================================================
#
# Cada linea:   sensor=fuente,parametro=valor,parametro=valor...
# Lo que va despues de # es un comentario y no se lee.
#
# Sensores:  rpm, velocidad, turbo, agua, aceite, pres_aceite, bateria, afr,
#            pres_nafta, admision, tps, egt
#
# Fuentes:
#   apagado    no se usa (es lo mismo que no ponerlo)
#   sim        valores de prueba
#   analog     sensor analogico en una entrada del ADC
#   pulso      cuenta pulsos: bobina / tacometro (rpm) o sensor VSS (velocidad)
#   ecu        lo manda la ECU por CAN / OBD-II (todavia en desarrollo)
#
# Los sensores que no aparecen quedan apagados, salvo con:
simular=no            # si = los que no estan en la lista salen simulados
#
# ---------------------------------------------------------------------
#  Entradas del ESP32-P4
# ---------------------------------------------------------------------
#  Analogicas (ADC): GPIO20 a 23 y GPIO49 a 52. (GPIO16 a 19 tambien tienen ADC,
#  pero en esta placa pueden estar conectados al chip de WiFi.)
#  Pulsos: cualquier GPIO libre.
#  La placa ya usa: 7-13, 24-27, 32, 33, 37-44, 53 (la pantalla avisa si se elige uno de esos).
#
#  IMPORTANTE: los pines aguantan como maximo 3,3 V.
#  - Sensores de 5 V: poner un divisor (por ejemplo 10k arriba y 20k abajo) y
#    escribir divisor= con la relacion (con 10k/20k es 1.5).
#  - Bateria: divisor de 100k arriba y 10k abajo (divisor=11, es el de fabrica).
#  - RPM desde la bobina: SIEMPRE con optoacoplador o un modulo de tacometro.
#    El negativo de bobina tiene picos de cientos de voltios.
#
divisor=1.0           # divisor de todas las entradas analogicas (cada sensor puede tener el suyo)
#
# ---------------------------------------------------------------------
#  Tipos de sensor (tipo=)
# ---------------------------------------------------------------------
#  pres_100psi, pres_150psi   presion 0,5-4,5 V (aceite, nafta)
#  pres_10bar                 presion 0,5-4,5 V, 0-10 bar
#  map_2bar, map_3bar, map_4bar   MAP 0,5-4,5 V (turbo; se muestra 0 = presion atmosferica)
#  wideband_10_20             banda ancha 0-5 V = 10-20 AFR
#  wideband_aem_x             AEM X-Series: 0,5-4,5 V = 8,5-18 AFR
#  wideband_innovate          Innovate LC-2 de fabrica: 0-5 V = 7,35-22,39 AFR
#  egt_5mv                    amplificador de termocupla 5 mV/°C
#  ntc_bosch, ntc_gm          temperatura (agua, aceite, admision)
#  ntc                        temperatura con otra NTC: r25= (ohms a 25 °C) y beta=
#  volts                      voltaje directo (bateria)
#  lineal                     cualquier otro: v1=,val1=,v2=,val2= (dos puntos voltaje -> valor)
#
#  Las presiones se convierten solas a la unidad de la pantalla
#  (turbo en psi, aceite y nafta en bar).
#
#  Sensores de temperatura (NTC): van a masa por un lado, y por el otro
#  al pin con una resistencia de pull-up a 3,3 V.
#    pullup=2200     valor de la resistencia de pull-up (ohms)
#    vpullup=3.3     a que voltaje va la pull-up
#
#  filtro=0.3        suavizado: 1 = sin filtro, 0.05 = mucho (de fabrica 0.05 en
#                    temperaturas, 0.1 en bateria y 0.3 en lo demas)
#
# ---------------------------------------------------------------------
#  Pulsos
# ---------------------------------------------------------------------
#  cilindros=4         motor 4 tiempos con una chispa por cilindro (distribuidor)
#  pulsos_vuelta=2     o directamente los pulsos por vuelta del cigueñal
#  pulsos_km=4000      velocidad: pulsos del VSS por kilometro
#  flanco=bajada       bajada (de fabrica, optoacoplador a masa) o subida
#  min_us=200          pulsos mas cortos que esto se ignoran (ruido)
#
# =====================================================================
#  EJEMPLO: motor carburado de 4 cilindros con turbo
# =====================================================================
rpm=pulso,gpio=4,cilindros=4
velocidad=pulso,gpio=5,pulsos_km=4000
turbo=analog,gpio=20,tipo=map_3bar,divisor=1.5
agua=analog,gpio=21,tipo=ntc_bosch,pullup=2200
aceite=analog,gpio=22,tipo=ntc_bosch,pullup=2200
pres_aceite=analog,gpio=23,tipo=pres_100psi,divisor=1.5
pres_nafta=analog,gpio=49,tipo=pres_100psi,divisor=1.5
afr=analog,gpio=50,tipo=wideband_aem_x,divisor=1.5
bateria=analog,gpio=51,tipo=volts,divisor=11

# egt=analog,gpio=52,tipo=egt_5mv,divisor=1.5
#

# rpm=ecu
# agua=ecu
